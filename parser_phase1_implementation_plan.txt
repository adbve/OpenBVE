public static class StreamingLineReader
{
    // Reusable buffer - 64KB default
    private const int DefaultBufferSize = 65536;
    
    /// <summary>
    /// Enumerates lines from a file without loading entire file into memory
    /// Uses ArrayPool for buffer to reduce allocations
    /// </summary>
    public static IEnumerable<(string Line, int LineNumber)> ReadLines(
        string filePath, 
        Encoding encoding = null,
        int bufferSize = DefaultBufferSize)
    {
        encoding ??= TextEncoding.GetSystemEncodingFromFile(filePath);
        
        // Use ArrayPool for buffer - no allocation!
        char[] buffer = System.Buffers.ArrayPool<char>.Shared.Rent(bufferSize);
        
        try
        {
            using var stream = new FileStream(
                filePath, 
                FileMode.Open, 
                FileAccess.Read, 
                FileShare.Read,
                bufferSize,
                FileOptions.SequentialScan); // Hint to OS for better perf
            
            using var reader = new StreamReader(stream, encoding, true, buffer.Length);
            
            int lineNumber = 0;
            string line;
            
            while ((line = await reader.ReadLineAsync()) != null)
            {
                lineNumber++;
                yield return (line, lineNumber);
            }
        }
        finally
        {
            // Return buffer to pool - CRITICAL!
            System.Buffers.ArrayPool<char>.Shared.Return(buffer);
        }
    }
    
    /// <summary>
    /// Synchronous version for contexts where async is not available
    /// </summary>
    public static IEnumerable<(string Line, int LineNumber)> ReadLinesSync(
        string filePath, 
        Encoding encoding = null,
        int bufferSize = DefaultBufferSize)
    {
        // Similar implementation but synchronous
    }
}

public static class SpanStringUtilities
{
    /// <summary>
    /// Splits a string without allocations using Span<char>
    /// Returns enumerator for efficient iteration
    /// </summary>
    public static SpanEnumerator Split(ReadOnlySpan<char> input, char separator)
    {
        return new SpanEnumerator(input, separator);
    }
    
    public ref struct SpanEnumerator
    {
        private readonly ReadOnlySpan<char> _input;
        private readonly char _separator;
        private int _currentIndex;
        private ReadOnlySpan<char> _current;
        
        public SpanEnumerator(ReadOnlySpan<char> input, char separator)
        {
            _input = input;
            _separator = separator;
            _currentIndex = 0;
            _current = ReadOnlySpan<char>.Empty;
        }
        
        public bool MoveNext()
        {
            if (_currentIndex > _input.Length)
                return false;
                
            int separatorIndex = _input.Slice(_currentIndex).IndexOf(_separator);
            
            if (separatorIndex == -1)
            {
                _current = _input.Slice(_currentIndex);
                _currentIndex = _input.Length + 1;
            }
            else
            {
                _current = _input.Slice(_currentIndex, separatorIndex);
                _currentIndex += separatorIndex + 1;
            }
            
            return true;
        }
        
        public ReadOnlySpan<char> Current => _current;
    }
    
    /// <summary>
    /// Parse double from Span<char> without string allocation
    /// </summary>
    public static bool TryParseDouble(ReadOnlySpan<char> span, out double result)
    {
        // Use Utf8Parser for ASCII, or fallback to double.Parse with stackalloc
        if (Utf8Parser.TryParse(span, out result, out _))
            return true;
            
        // Fallback for non-ASCII
        Span<char> buffer = stackalloc char[span.Length];
        span.CopyTo(buffer);
        return double.TryParse(buffer.ToString(), NumberStyles.Any, 
            CultureInfo.InvariantCulture, out result);
    }
    
    /// <summary>
    /// Parse int from Span<char> without string allocation  
    /// </summary>
    public static bool TryParseInt(ReadOnlySpan<char> span, out int result)
    {
        if (Utf8Parser.TryParse(span, out result, out _))
            return true;
            
        Span<char> buffer = stackalloc char[span.Length];
        span.CopyTo(buffer);
        return int.TryParse(buffer.ToString(), NumberStyles.Any, 
            CultureInfo.InvariantCulture, out result);
    }
    
    /// <summary>
    /// Trim whitespace from Span<char> without allocation
    /// </summary>
    public static ReadOnlySpan<char> Trim(this ReadOnlySpan<char> span)
    {
        int start = 0;
        while (start < span.Length && char.IsWhiteSpace(span[start]))
            start++;
            
        int end = span.Length;
        while (end > start && char.IsWhiteSpace(span[end - 1]))
            end--;
            
        return span.Slice(start, end - start);
    }
}

public static class PooledBufferManager
{
    // Shared pools for common sizes
    private static readonly ConcurrentBag<string[]> SmallStringArrays 
        = new ConcurrentBag<string[]>();
    
    private const int SmallArraySize = 16;
    private const int MediumArraySize = 256;
    private const int LargeArraySize = 4096;
    
    /// <summary>
    /// Rent a string array from pool
    /// </summary>
    public static string[] RentStringArray(int minimumLength)
    {
        if (minimumLength <= SmallArraySize && SmallStringArrays.TryTake(out var arr))
        {
            return arr;
        }
        
        // Fall back to ArrayPool for other types/sizes
        return System.Buffers.ArrayPool<string>.Shared.Rent(minimumLength);
    }
    
    /// <summary>
    /// Return array to pool - MUST be called after use
    /// </summary>
    public static void ReturnStringArray(string[] array)
    {
        if (array.Length == SmallArraySize)
        {
            // Clear before returning to prevent memory leaks
            Array.Clear(array, 0, array.Length);
            SmallStringArrays.Add(array);
        }
        else
        {
            System.Buffers.ArrayPool<string>.Shared.Return(array, clearArray: true);
        }
    }
    
    /// <summary>
    /// Helper for using pooled arrays with automatic cleanup
    /// </summary>
    public static PooledArrayHandle UsePooledStringArray(int minimumLength)
    {
        return new PooledArrayHandle(RentStringArray(minimumLength));
    }
}

public sealed class PooledArrayHandle : IDisposable
{
    public string[] Array { get; }
    private bool _disposed;
    
    public PooledArrayHandle(string[] array)
    {
        Array = array;
    }
    
    public void Dispose()
    {
        if (!_disposed)
        {
            PooledBufferManager.ReturnStringArray(Array);
            _disposed = true;
        }
    }
}

USAGE PATTERN:
    using (var handle = PooledBufferManager.UsePooledStringArray(16))
    {
        string[] args = handle.Array;
        // Use args...
    } // Automatically returned to pool
